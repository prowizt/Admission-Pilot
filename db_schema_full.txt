=========================================
Table: Sys_AIAuditLog (AI 챗봇 대화 및 모니터링 로그)
=========================================
Columns:
 - [PK] id (int(4)) NOT NULL
 - user_role (varchar(50)) NULL -- 사용자 직군(부서)
 - model_name (varchar(100)) NULL -- 사용된 AI 모델명
 - question (nvarchar) NULL -- 사용자 실제 질문 원문
 - scraped_context (nvarchar) NULL -- 참고한 웹/스크랩 텍스트
 - sql_query (nvarchar) NULL -- 생성된 T-SQL 쿼리 (없으면 NONE)
 - rag_query (nvarchar) NULL -- RAG 문서 검색 키워드
 - answer (nvarchar) NULL -- AI가 응답한 최종 답변
 - latency_ms (int(4)) NULL -- 응답 소요 시간(ms)
 - created_at (datetime(8)) NULL -- 로그 생성 일시
 - user_feedback (varchar(10)) NULL -- AI 답변 피드백 (UP: 좋아요, DOWN: 싫어요)

=========================================
Table: Sys_ColumnCatalog (컬럼 레벨 보안(CLS) 및 AI 힌트 카탈로그)
=========================================
Columns:
 - [PK] id (int(4)) NOT NULL -- 고유 ID
 - table_name (nvarchar(200)) NOT NULL -- 테이블명
 - column_name (nvarchar(200)) NOT NULL -- 컬럼명
 - ai_description (nvarchar) NULL -- AI 설명 힌트 (예: 이 컬럼은 학번)
 - is_public (char(1)) NULL -- 공개 여부 (Y: 전체 공개, N: 직원 전용)
 - created_at (datetime(8)) NULL -- 등록일시

=========================================
Table: Sys_DocumentCatalog (비정형 지식(PDF) 문서 카탈로그)
=========================================
Columns:
 - [PK] doc_id (nvarchar(200)) NOT NULL -- 문서 고유 ID (ChromaDB UUID와 동일)
 - filename (nvarchar(510)) NOT NULL -- 파일명
 - doc_type (nvarchar(100)) NOT NULL -- 문서 유형 (rule 또는 reference)
 - year (nvarchar(20)) NULL -- 적용년도
 - title (nvarchar(510)) NULL -- 문서 제목
 - is_public (char(1)) NULL -- 문서 공개 여부 (Y: 공개, N: 직원 전용)
 - uploaded_at (datetime(8)) NULL -- 업로드 일시
 - description (nvarchar) NULL -- 문서 설명 및 AI/관리자 힌트

=========================================
Table: Sys_SupplementalKnowledge (실시간 보완 지식 및 인사/결재 예외 장부 테이블)
=========================================
Columns:
 - [PK] id (int(4)) NOT NULL -- 고유 일련번호 (PK)
 - category (nvarchar(200)) NOT NULL -- 지식 분류 (예: 인사/조직, 결재라인, 예외규정 등)
 - content (nvarchar(2000)) NOT NULL -- 실시간 사전지식 팩트 원문
 - is_active (char(1)) NOT NULL -- 활성 여부 (Y: 활성, N: 비활성)
 - created_at (datetime(8)) NOT NULL -- 등록/수정 일시
 - author (nvarchar(200)) NULL -- 등록한 교직원명 또는 ID

=========================================
Table: Sys_TableCatalog (테이블 관리 카탈로그 (외부/내부 DB 출처 관리))
=========================================
Columns:

 - db_source (nvarchar(100)) NULL -- DB 출처 구분 (INTERNAL: 엑셀, EXTERNAL: 타 시스템)
 - description (nvarchar) NULL -- 테이블 용도 설명
 - created_at (datetime(8)) NULL -- 등록일시
